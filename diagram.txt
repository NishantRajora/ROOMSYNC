database "SQLite3 (db.sqlite3)" {
    entity "roomsync_accounts" as TableAccounts {
        * id : INT (PK)
        --
        email : TEXT (Unique)
        password_hash : TEXT
        budget : TEXT
        sleep/cleanliness : TEXT
    }
    entity "roomsync_conversations" as TableConvs {
        * id : INT (PK)
        --
        participant_1 : TEXT
        participant_2 : TEXT
        listing_id : INT
    }
    entity "roomsync_messages" as TableMsgs {
        * id : INT (PK)
        --
        conversation_id : INT (FK)
        sender : TEXT
        text : TEXT
        read : INT
    }
    entity "roomsync_expenses" as TableExp {
        * id : INT (PK)
        --
        group_id : TEXT
        amount : REAL
        paid_by : TEXT
        upi_id : TEXT
    }
    entity "roomsync_pacts" as TablePacts {
        * id : INT (PK)
        --
        agreement_text : TEXT
        sha256_hash : TEXT
        tx_hash : TEXT
    }
    entity "roomsync_reviews" as TableReviews {
        * id : INT (PK)
        --
        locality : TEXT
        overall_rating : INT
        deposit_returned : INT
    }
    entity "roomsync_visit_alerts" as TableAlerts {
        * id : INT (PK)
        --
        user_email : TEXT
        status : TEXT (active/safe/SOS)
    }
    entity "roomsync_student_verifications" as TableVerify {
        * id : INT (PK)
        --
        email : TEXT (Unique)
        otp : TEXT
        verified : INT
    }
}

cloud "Blockchain (Ethereum/Polygon)" {
    component [AgreementRegistry.sol] as SmartContract <<Smart Contract>>
}

' --- Frontend to Backend Interactions ---
Discovery --> CoreAPI : GET /api/listings/ (Search/Filter)
Discovery --> CoreAPI : POST /api/listings/ (Create)
Discovery --> CoreAPI : POST /api/listings/{id}/connect/ (Connect)
Chat --> ChatAPI : GET /api/chat/conversations/
Chat --> ChatAPI : POST /api/chat/messages/
ChatPopup --> ChatAPI : POST /api/chat/conversations/ (Init)
AgreementVerifier --> CoreAPI : POST /api/agreements/analyse/
ExpenseSplitter --> CommunityAPI : GET /api/expenses/balances/
RoommatePact --> CommunityAPI : POST /api/pacts/ (Gen Agreement)
FlatVisit --> CommunityAPI : POST /api/safety/visit-alerts/ (Start/SOS)
SocietyReviews --> CommunityAPI : GET/POST /api/reviews/
StudentVerify --> CommunityAPI : POST /api/auth/student-otp/
AdminDB --> CoreAPI : GET /api/admin/database/

' --- Backend Internal Logic ---
CoreAPI --> MatchEngine : rank_matches(profile, candidates)
CoreAPI --> TrustEngine : listing_trust(listing)
CoreAPI --> SafetyEngine : safety_report(lat, lng)
CoreAPI --> NLPAnalyser : analyse_agreement(text)
CoreAPI --> EnhancedNLP : summarize_agreement(text)

' --- Data Persistence ---
CoreAPI ..> TableAccounts
ChatAPI ..> TableConvs
ChatAPI ..> TableMsgs
CommunityAPI ..> TableExp
CommunityAPI ..> TablePacts
CommunityAPI ..> TableReviews
CommunityAPI ..> TableAlerts
CommunityAPI ..> TableVerify

' --- Blockchain Integration ---
AgreementVerifier --> BC_Client : registerHash(sha256)
AgreementVerifier --> BC_Client : verifyHash(sha256)
RoommatePact --> BC_Client : registerHash(sha256)
BC_Client --> SmartContract : register(bytes32)
BC_Client --> SmartContract : get(bytes32)
